if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6TinyRockSmashNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_TINY_ROCKSMASH_SOURCES bridge/gen6tinyrocksmash_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6tinyrocksmash ${GEN6_TINY_ROCKSMASH_SOURCES})
    target_include_directories(gen6tinyrocksmash PRIVATE bridge)
    target_compile_options(gen6tinyrocksmash PRIVATE "-O3")
    target_link_options(gen6tinyrocksmash PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGen6TinyRockSmashModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen6tinyrocksmash_api_version','_gen6tinyrocksmash_begin','_gen6tinyrocksmash_step','_gen6tinyrocksmash_result_ptr','_gen6tinyrocksmash_result_count','_gen6tinyrocksmash_step_processed','_gen6tinyrocksmash_total_processed','_gen6tinyrocksmash_done','_gen6tinyrocksmash_limit_reached','_gen6tinyrocksmash_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6tinyrocksmash PROPERTIES OUTPUT_NAME "gen6tinyrocksmash" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6tinyrocksmash_native_test ${GEN6_TINY_ROCKSMASH_SOURCES} tests/gen6tinyrocksmash_native_test.cpp)
    target_include_directories(gen6tinyrocksmash_native_test PRIVATE bridge)
    add_test(NAME gen6tinyrocksmash_native_parity COMMAND gen6tinyrocksmash_native_test)
endif()
