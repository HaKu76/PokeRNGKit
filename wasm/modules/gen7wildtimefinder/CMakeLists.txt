    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen7WildTimeFinderNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)

set(GEN7_WILD_TIME_FINDER_SOURCES bridge/gen7wildtimefinder_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen7wildtimefinder ${GEN7_WILD_TIME_FINDER_SOURCES})
    target_include_directories(gen7wildtimefinder PRIVATE bridge)
    target_compile_options(gen7wildtimefinder PRIVATE "-O3")
    target_link_options(gen7wildtimefinder PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGen7WildTimeFinderModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen7wildtimefinder_api_version','_gen7wildtimefinder_begin','_gen7wildtimefinder_step','_gen7wildtimefinder_result_ptr','_gen7wildtimefinder_result_count','_gen7wildtimefinder_step_processed','_gen7wildtimefinder_total_processed','_gen7wildtimefinder_total_results','_gen7wildtimefinder_done','_gen7wildtimefinder_limit_reached','_gen7wildtimefinder_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen7wildtimefinder PROPERTIES OUTPUT_NAME "gen7wildtimefinder" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen7wildtimefinder_native_test ${GEN7_WILD_TIME_FINDER_SOURCES} tests/gen7wildtimefinder_native_test.cpp)
    target_include_directories(gen7wildtimefinder_native_test PRIVATE bridge)
    add_test(NAME gen7wildtimefinder_session COMMAND gen7wildtimefinder_native_test)
endif()
