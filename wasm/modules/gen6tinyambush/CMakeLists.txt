if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6TinyAmbushNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_TINY_AMBUSH_SOURCES bridge/gen6tinyambush_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6tinyambush ${GEN6_TINY_AMBUSH_SOURCES})
    target_include_directories(gen6tinyambush PRIVATE bridge)
    target_compile_options(gen6tinyambush PRIVATE "-O3")
    target_link_options(gen6tinyambush PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=creategen6tinyambushModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen6tinyambush_api_version','_gen6tinyambush_begin','_gen6tinyambush_step','_gen6tinyambush_result_ptr','_gen6tinyambush_result_count','_gen6tinyambush_step_processed','_gen6tinyambush_total_processed','_gen6tinyambush_done','_gen6tinyambush_limit_reached','_gen6tinyambush_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6tinyambush PROPERTIES OUTPUT_NAME "gen6tinyambush" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6tinyambush_native_test ${GEN6_TINY_AMBUSH_SOURCES} tests/gen6tinyambush_native_test.cpp)
    target_include_directories(gen6tinyambush_native_test PRIVATE bridge)
    add_test(NAME gen6tinyambush_native_parity COMMAND gen6tinyambush_native_test)
endif()
