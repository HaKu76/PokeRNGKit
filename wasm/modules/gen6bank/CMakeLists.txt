if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6BankNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_BANK_SOURCES bridge/gen6bank_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6bank ${GEN6_BANK_SOURCES})
    target_include_directories(gen6bank PRIVATE bridge)
    target_compile_options(gen6bank PRIVATE "-O3")
    target_link_options(gen6bank PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGen6BankModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen6bank_api_version','_gen6bank_generate','_gen6bank_result_ptr','_gen6bank_result_count','_gen6bank_processed_count','_gen6bank_limit_reached','_gen6bank_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6bank PROPERTIES OUTPUT_NAME "gen6bank" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6bank_native_test ${GEN6_BANK_SOURCES} tests/gen6bank_native_test.cpp)
    target_include_directories(gen6bank_native_test PRIVATE bridge)
    add_test(NAME gen6bank_native_parity COMMAND gen6bank_native_test)
endif()
