if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6WildNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_WILD_SOURCES bridge/gen6wild_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6wild ${GEN6_WILD_SOURCES})
    target_include_directories(gen6wild PRIVATE bridge)
    target_compile_options(gen6wild PRIVATE "-O3")
    target_link_options(gen6wild PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGen6WildModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen6wild_api_version','_gen6wild_generate','_gen6wild_result_ptr','_gen6wild_result_count','_gen6wild_processed_count','_gen6wild_limit_reached','_gen6wild_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6wild PROPERTIES OUTPUT_NAME "gen6wild" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6wild_native_test ${GEN6_WILD_SOURCES} tests/gen6wild_native_test.cpp)
    target_include_directories(gen6wild_native_test PRIVATE bridge)
    add_test(NAME gen6wild_native_parity COMMAND gen6wild_native_test)
endif()
