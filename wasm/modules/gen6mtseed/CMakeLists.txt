if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6MtSeedNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_MT_SEED_SOURCES bridge/gen6mtseed_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6mtseed ${GEN6_MT_SEED_SOURCES})
    target_include_directories(gen6mtseed PRIVATE bridge)
    target_compile_options(gen6mtseed PRIVATE "-O3")
    target_link_options(gen6mtseed PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGen6MtSeedModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_gen6mtseed_api_version','_gen6mtseed_begin','_gen6mtseed_step','_gen6mtseed_result_ptr','_gen6mtseed_result_count','_gen6mtseed_step_processed','_gen6mtseed_total_processed','_gen6mtseed_done','_gen6mtseed_limit_reached','_gen6mtseed_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6mtseed PROPERTIES OUTPUT_NAME "gen6mtseed" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6mtseed_native_test ${GEN6_MT_SEED_SOURCES} tests/gen6mtseed_native_test.cpp)
    target_include_directories(gen6mtseed_native_test PRIVATE bridge)
    add_test(NAME gen6mtseed_native_parity COMMAND gen6mtseed_native_test)
endif()
