if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.31)
    project(PokeRNGKitGen6TinyHoneyNative LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(CTest)
endif()

set(GEN6_TINY_HONEY_SOURCES bridge/gen6tinyhoney_bridge.cpp)

if(EMSCRIPTEN)
    add_executable(gen6tinyhoney ${GEN6_TINY_HONEY_SOURCES})
    target_include_directories(gen6tinyhoney PRIVATE bridge)
    target_compile_options(gen6tinyhoney PRIVATE "-O3")
    target_link_options(gen6tinyhoney PRIVATE "-O3" "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=creategen6tinyhoneyModule" "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFILESYSTEM=0" "-sASSERTIONS=0" "-sNO_EXIT_RUNTIME=1" "-sMALLOC=emmalloc" "-sEXPORTED_FUNCTIONS=['_malloc','_free','_gen6tinyhoney_api_version','_gen6tinyhoney_begin','_gen6tinyhoney_step','_gen6tinyhoney_result_ptr','_gen6tinyhoney_result_count','_gen6tinyhoney_step_processed','_gen6tinyhoney_total_processed','_gen6tinyhoney_done','_gen6tinyhoney_limit_reached','_gen6tinyhoney_last_error']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32']")
    set_target_properties(gen6tinyhoney PROPERTIES OUTPUT_NAME "gen6tinyhoney" SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${POKERNGKIT_WASM_OUTPUT_DIR}")
elseif(BUILD_TESTING)
    add_executable(gen6tinyhoney_native_test ${GEN6_TINY_HONEY_SOURCES} tests/gen6tinyhoney_native_test.cpp)
    target_include_directories(gen6tinyhoney_native_test PRIVATE bridge)
    add_test(NAME gen6tinyhoney_native_parity COMMAND gen6tinyhoney_native_test)
endif()
